```cpp
    #include <string>

    namespace asio = boost::asio;
    namespace beast = boost::beast;

    namespace websocket = beast::websocket;

    // Namespace for your engine
    namespace Anomalon {

        // Simple state shared with Lua for dynamic updates
        struct EngineState {

            std::string server_status = "Operational";
        };

        // Coroutine to handle an established SSL WebSocket session
        asio::awaitable<void> run_websocket_session(beast::ssl_stream<beast::tcp_stream> stream, std::shared_ptr<sol::state> lua) {
            try {
                websocket::stream<beast::ssl_stream<beast::tcp_stream>&> ws{stream};

                // Set suggested options for the websocket
                ws.set_option(websocket::stream_base::timeout::suggested(beast::role_type::server));

                // Accept the websocket handshake
                co_await ws.async_accept(asio::use_awaitable);

                beast::flat_buffer buffer;
                while(true) {
                    // Read a message
                    co_await ws.async_read(buffer, asio::use_awaitable);

                    // Call out to Lua to process the incoming payload dynamically
                    std::string input = beast::buffers_to_string(buffer.data());
                    std::string response = (*lua)["on_websocket_message"](input);

                    // Echo or send the updated payload back to client
                    ws.text(ws.got_text());
                    co_await ws.async_write(asio::buffer(response), asio::use_awaitable);
                    buffer.consume(buffer.size());
                }
            } catch (const boost::system::system_error& se) {
                if (se.code() != websocket::error::closed)
                    std::cerr << "[WS Error] " << se.what() << "\n";
            }
        }

        // Coroutine to handle an SSL HTTP session, upgrading to WebSockets if requested
        asio::awaitable<void> handle_session(tcp::socket socket, asio::ssl::context& ctx, std::shared_ptr<sol::state> lua) {
            beast::ssl_stream<beast::tcp_stream> stream{std::move(socket), ctx};
            beast::flat_buffer buffer;

            // Perform SSL Handshake
            co_await stream.async_handshake(asio::ssl::stream_base::server, asio::use_awaitable);

            while(true) {
                http::request<http::string_body> req;
                // Read HTTP request
                co_await http::async_read(stream, buffer, req, asio::use_awaitable);

                // Check if it's a WebSocket Upgrade request
                if(websocket::is_upgrade(req)) {
                    // Relinquish control to the WebSocket coroutine loop
                    co_await run_websocket_session(std::move(stream), lua);
                    co_return;
                }

                // Route standard HTTPS traffic using Lua script logic
                std::string path = std::string(req.target());
                sol::protected_function handler = (*lua)["route_http"];

                std::string body = "404 Not Found";
                http::status status = http::status::not_found;

                if (handler.valid()) {
                    sol::protected_function_result result = handler(path);
                    if (result.valid()) {
                        body = result.get<std::string>();
                        status = http::status::ok;
                    }
                }

                // Assemble and send HTTPS response
                http::response<http::string_body> res{status, req.version()};
                res.set(http::field::server, "Anomalon Core/1.0");
                res.set(http::field::content_type, "text/html");
                res.keep_alive(req.keep_alive());
                res.body() = body;
                res.prepare_payload();

                co_await http::async_write(stream, res, asio::use_awaitable);

                if(!req.keep_alive()) break;
            }

            // Gracefully shut down SSL connection
            beast::error_code ec;
            co_await stream.async_shutdown(asio::use_awaitable);
        }

        // Coroutine listener that accepts incoming TCP connections
        asio::awaitable<void> listener(unsigned short port, asio::ssl::context& ctx, std::shared_ptr<sol::state> lua) {
            auto executor = co_await asio::this_coro::executor;
            tcp::acceptor acceptor{executor, {tcp::v4(), port}};

            std::cout << "[Anomalon] Serving secure traffic on port " << port << "...\n";

            while(true) {
                tcp::socket socket = co_await acceptor.async_accept(asio::use_awaitable);
                // Spawn a new detached coroutine for every unique session safely
                asio::co_spawn(executor, handle_session(std::move(socket), ctx, lua), asio::detached);
            }
        }
    }

        try {

            // Initialize SSL Context
            asio::ssl::context ctx{asio::ssl::context::tlsv13};
            ctx.use_certificate_chain_file("server.crt");
```

# if ios is provided with multiple threads, is the variable ua thread safe



No, the std::shared_ptr<sol::state> lua variable (and the Lua state itself) is absolutely not thread-safe.

If your Boost.Asio io_context runs on a multi-threaded pool (e.g., calling ioc.run() from multiple threads), passing and using that single lua pointer across concurrent coroutines will cause race conditions, memory corruption, and crashes.

## Why it fails in a multi-threaded environment

   1. Lua is inherently single-threaded: The standard Lua runtime is strictly non-reentrant. It relies on internal global states. Two threads trying to read or write to the same Lua state simultaneously will corrupt the Lua stack.
   2. sol2 does not auto-lock: While sol2 provides abstract wrappers like sol::state and sol::protected_function, it does not add an internal mutex to thread-protect calls.
   3. std::shared_ptr thread-safety traps: While copying a std::shared_ptr itself is thread-safe (the reference counter is atomic), modifying or concurrent accessing the underlying object it points to is not.

## How to fix it for Multi-threaded io_context

You have two architectural options to make Anomalon thread-safe while using multiple threads:

### Option A: One Lua State per Thread (Recommended for Performance)

Instead of a single global Lua state, give each thread its own completely independent sol::state. Because there is no shared state, there is no lock contention, allowing you to maximize C++20 coroutine speeds.

You can use thread_local to manage this smoothly:
```cpp
    // Inside main.cpp or engine class
    thread_local std::unique_ptr<sol::state> local_lua = nullptr;

    sol::state& get_lua_instance() {
        if (!local_lua) {
            local_lua = std::make_unique<sol::state>();
            local_lua->open_libraries(sol::lib::base, sol::lib::string);
            // Bind your C++ types to this specific thread instance here...
            local_lua->script_file("scripts/server.lua");
        }
        return *local_lua;
    }
```
Then, inside your coroutine sessions (handle_session), fetch the instance safely without passing a shared pointer:
```cpp
    // Grab the safe, isolated instance for the current executing thread
    auto& lua = get_lua_instance();
    std::string response = lua["on_websocket_message"](input);
```

### Option B: Strand / Mutex Locking (If states MUST be shared)

If your Lua state absolutely has to share mutable data globally across requests, you must enforce mutual exclusion using an Asio strand or a std::mutex.

```cpp
    // A wrap structure protecting the state
    struct ProtectedLua {
        std::mutex mtx;
        sol::state lua;
    };

    // Inside your thread coroutine:
    std::string response;
    {
        std::lock_guard<std::mutex> lock(protected_lua->mtx);
        response = protected_lua->lua["on_websocket_message"](input);
    } // Lock releases immediately after the execution finishes
```
Note: This creates a performance bottleneck since all your high-speed coroutines will queue up waiting for the Lua lock.

# can there be multiple co_spawn instances per thread

Yes, absolutely. You can have thousands of active co_spawn coroutines running concurrently on a single OS thread.
This is the entire superpower of Boost.Asio and C++20 coroutines. It is a one-to-many architecture (one OS thread running many collaborative coroutines).

## How it works under the hood
When you call asio::co_spawn, you are creating a lightweight cooperative task (a fiber or green thread), not an operating system thread.

* Asio assigns these tasks to its internal event queue.
* The single OS thread calls ioc.run(), which acts as a fast loop processing that queue.
* When a coroutine hits a co_await statement (like waiting for async_read), it completely pauses execution and yields control.
* Instead of blocking the OS thread, the thread immediately jumps to the next available coroutine task in the queue.


## The Thread-Local Isolation Impact
Because you chose Option A, you need to understand how thread_local behaves when multiple coroutines share the same thread:

| Scenario | Behavior | Thread Safety Status |
|---|---|---|
| Coroutine A and Coroutine B are on different OS threads | They use entirely different sol::state instances. | 100% Safe (No sharing) |
| Coroutine A and Coroutine B are on the exact same OS thread | They share the same thread-local sol::state instance. | Safe (Sequential execution) |

## Why sharing a Lua state via thread_local is safe:
Even though Coroutine A and Coroutine B on the same thread are touching the exact same sol::state, they cannot execute at the exact same millisecond.

Because a single CPU thread can only execute one line of assembly code at a time, Coroutine A must completely yield control (via co_await) before Coroutine B can wake up and execute. There is no structural way for them to corrupt the Lua stack simultaneously.

## The One Critical Rule for Lua State Management

Because coroutines on the same thread share that thread's local Lua instance, they share the top-level global variables in that specific Lua script.
If Coroutine A sets a global variable in Lua:
```lua
    -- Inside scripts/server.lua
    function on_websocket_message(payload)
        my_global_var = payload -- DANGEROUS IF SHARED
    end
```
And then Coroutine A pauses (co_await), Coroutine B could wake up on that same thread and overwrite my_global_var before Coroutine A wakes back up to finish its work.

The Fix: Always use the local keyword inside your Lua scripts for request-scoped data:
```lua
    -- scripts/server.lua
    function on_websocket_message(payload)
        local safe_request_data = payload -- Perfectly safe across thousands of concurrent coroutines
        return "Processed: " .. safe_request_data
    end
```

# Will coroutine A always be processed on the thread it started on

Not necessarily. Whether Coroutine A stays on the same thread depends entirely on how you configure your Boost.Asio Executor and Strand.

By default, in the multi-threaded code we wrote above, coroutines can hop between threads whenever they wake up from a co_await boundary.

## Why Coroutines Hop Threads

When you call ioc.run() on multiple threads, they all pull tasks from a single, shared global queue.

   1. Coroutine A starts executing on Thread 1.
   2. It hits co_await http::async_read(...) and pauses.
   3. The network read finishes. The OS signals Asio that data is ready.
   4. Asio pushes the resumption task back onto the global queue.
   5. If Thread 2 happens to be free at that exact millisecond, it will grab Coroutine A from the queue and finish executing it.

## The Problem: This Breaks thread_local
Because Coroutine A can start on Thread 1 and wake up on Thread 2, relying on thread_local inside the coroutine becomes dangerous. Coroutine A might modify a thread-local variable on Thread 1, pause, wake up on Thread 2, and read a completely different thread-local variable.

## The Fix: Force Thread Affinity Using asio::strand
To guarantee that a coroutine always stays on the exact thread it started on, you must bind its execution to an asio::strand. A strand enforces sequential execution and ensures that all tasks submitted to it run on a single thread.

Here is how you update the Anomalon architecture to pin coroutines to their originating thread:

## 1. Update the Listener to use Strands
When a new connection arrives, allocate a dedicated strand for that session before spawning the coroutine:
```cpp
    // Inside Anomalon::listener coroutine
    while(true) {
        tcp::socket socket = co_await acceptor.async_accept(asio::use_awaitable);

        // Create a strand specifically for this socket's executor
        auto session_strand = asio::make_strand(acceptor.get_executor());

        // Spawn the coroutine bound strictly to this strand
        asio::co_spawn(session_strand, handle_session(std::move(socket), ctx), asio::detached);
    }
```

## 2. Ensure Async Operations Respect the Strand
Inside your session handlers, always use asio::bind_executor or ensure your streams are constructed using the socket's strand executor. When using beast::tcp_stream, passing the socket handles this automatically:
```cpp
    // Inside handle_session
    // beast::tcp_stream automatically adopts the strand executor bound to the moved socket
    beast::ssl_stream<beast::tcp_stream> stream{std::move(socket), ctx};
```
## Summary of Guarantees

* Without a Strand: Coroutine A will wake up on whichever thread is least busy. thread_local is unsafe here.
* With a Strand: Coroutine A is pinned to a single thread for its entire lifecycle. It can safely yield (co_await), let other coroutines run on that thread, and seamlessly resume on the same thread. thread_local is 100% safe here.

# What is the best way to maintain lua state per coroutine

The absolute best, most robust way to maintain a Lua state per coroutine in a multi-threaded Boost.Asio architecture is to use Lua Coroutines (also known as Threads in Lua, or lua_State* sub-states) managed via sol::thread.

Instead of assigning a massive, heavy, independent isolated sol::state to every connection, you maintain one main sol::state per OS thread (your global pool), and then create a lightweight sub-context for each C++ coroutine.

## Why this is the gold standard approach

   1. Memory Efficiency: A full sol::state carries massive overhead because it loads the entire base language libraries. A Lua thread (sub-state) shares the global environment memory (like your route functions) but keeps its own separate, isolated execution stack and local variable registers.
   2. Total Thread Safety: When paired with asio::strand (which pins your C++ coroutine to an OS thread), the Lua sub-state moves safely along with your C++ coroutine execution.
   3. No Cross-Talk: Coroutine A cannot leak state or variables into Coroutine B because their execution stacks are fully sandboxed from one another.


## Implementation Blueprint
Here is how you adjust Anomalon to implement Lua coroutines mapped 1:1 to C++ coroutines.

### 1. The Session Architecture
When a new connection is established, create a sol::thread inside your C++ session context. This pulls an isolated sub-state pointer (lua_State*) from your main thread-local engine pool.
```cpp
    #include <sol/sol.hpp>
    #include <boost/asio/co_spawn.hpp>
    #include <boost/asio/strand.hpp>

    namespace Anomalon {

        // Main thread-local runtime (one per OS thread)
        inline thread_local std::unique_ptr<sol::state> master_lua = nullptr;

        sol::state& get_master_lua() {
            if (!master_lua) {
                master_lua = std::make_unique<sol::state>();
                master_lua->open_libraries(sol::lib::base, sol::lib::string, sol::lib::table);
                master_lua->script_file("scripts/server.lua");
            }
            return *master_lua;
        }

        // C++ Coroutine representing a single connection session
        asio::awaitable<void> handle_session(tcp::socket socket, asio::ssl::context& ssl_ctx) {
            try {
                // 1. Fetch this thread's master state
                auto& master = get_master_lua();

                // 2. Create a lightweight Lua thread (sub-state) for THIS specific coroutine
                sol::thread lua_th = sol::thread::create(master);
                sol::state_view local_lua_state = lua_th.state();

                // 3. Initialize your SSL network stream
                beast::ssl_stream<beast::tcp_stream> stream{std::move(socket), ssl_ctx};
                beast::flat_buffer buffer;
                co_await stream.async_handshake(asio::ssl::stream_base::server, asio::use_awaitable);

                while(true) {
                    http::request<http::string_body> req;
                    co_await http::async_read(stream, buffer, req, asio::use_awaitable);

                    std::string path = std::string(req.target());

                    // 4. Safely execute the function inside this coroutine's personal sub-state view
                    sol::protected_function handler = local_lua_state["route_http"];
                    std::string body = "404 Not Found";

                    if (handler.valid()) {
                        auto result = co_await handler(path); // Safely executed on an isolated stack
                        if (result.valid()) {
                            body = result.get<std::string>();
                        }
                    }

                    // ... Assemble and write your HTTP/WebSocket response ...
                }
            } catch (const std::exception& e) {
                std::cerr << "[Session Error] " << e.what() << "\n";
            }
        }
    }
```

```cpp
    }

    // Network listener accepting connections and distributing directly to the global pool
    asio::awaitable<void> listener(unsigned short port, asio::ssl::context& ctx) {
        auto executor = co_await asio::this_coro::executor;
        tcp::acceptor acceptor{executor, {tcp::v4(), port}};

        std::cout << "[Anomalon] Statelessly balancing secure traffic on port " << port << "...\n";

        while(true) {
            tcp::socket socket = co_await acceptor.async_accept(asio::use_awaitable);

            // No strands used here. Spawning directly onto the global context executor
            // allows connections to aggressively steal and utilize any free worker thread.
            asio::co_spawn(executor, handle_session(std::move(socket), ctx), asio::detached);
        }
    }
}

int main() {
    try {
        asio::io_context ioc;

        asio::ssl::context ctx{asio::ssl::context::tlsv13};
        ctx.use_certificate_chain_file("server.crt");
        ctx.use_private_key_file("server.key", asio::ssl::context::pem);

        // Start the master listener
        asio::co_spawn(ioc, Anomalon::listener(443, ctx), asio::detached);

        // Spin up the hardware-optimized thread pool
        unsigned int thread_count = std::thread::hardware_concurrency();
        std::vector<std::thread> thread_pool;
        thread_pool.reserve(thread_count - 1);

        std::cout << "[Anomalon] Initializing stateless engine with pool size: " << thread_count << "\n";

        for(unsigned int i = 0; i < thread_count - 1; ++i) {
            thread_pool.emplace_back([&ioc]() {
                ioc.run();
            });
        }
        ioc.run();

        for(auto& t : thread_pool) {
            if(t.joinable()) t.join();
        }

    } catch (const std::exception& e) {
        std::cerr << "[Critical Exception] " << e.what() << "\n";
        return 1;
    }
    return 0;
}
```

## Why this design flies

* Work Stealing Efficiency: Because there are no strands pinning down paths, your OS thread pool can operate under maximum asynchronous concurrency. If one thread gets slowed down by an intensive computation, other threads will naturally steal pending socket notifications from the Asio queue.
* Lua Stack Virtualization: sol::thread::create(master) creates an entirely separate execution stack frame inside Lua. Even if an async request hops across 3 different physical CPU cores during its life cycle, its local variables inside Lua are perfectly safe and entirely independent.

This CMakeLists.txt handles a modern C++20 build. It automatically locates Boost (including its compiled components for Asio/Beast/SSL), OpenSSL, and sol2, while setting up proper optimization and warning flags.

## CMakeLists.txt
```cmake
cmake_minimum_required(VERSION 3.20)
project(Anomalon LANGUAGES CXX)

# Force modern C++20 standard requirements
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Determine the build mode (default to Release for max performance)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release")
endif()

# ---------------------------------------------------------------------------
# Find Required Packages
# ---------------------------------------------------------------------------

# 1. Find OpenSSL (Required for TLS/SSL encryption in Beast)
find_package(OpenSSL REQUIRED)

# 2. Find Boost (Beast is header-only, but its Asio backend needs compiled targets)
#    We require context/coroutine if your Asio layout needs deep stack traits,
#    but raw C++20 coroutines mostly rely on the core system threads.
find_package(Boost REQUIRED COMPONENTS system)

# 3. Find Lua (sol2 acts as the binder wrapping around the raw Lua C libraries)
find_package(Lua REQUIRED)

# 4. Find sol2 (Modern C++ Lua Binding Library)
#    If sol2 is installed via a package manager like vcpkg or apt, use find_package.
#    Alternatively, if you are treating sol2 as a header-only submodule,
#    you can point to its include path directly using target_include_directories instead.
find_package(sol2 CONFIG REQUIRED)

# ---------------------------------------------------------------------------
# Define Engine Target and Source Layout
# ---------------------------------------------------------------------------

add_executable(anomalon
    src/main.cpp
)

# ---------------------------------------------------------------------------
# Header Include Paths
# ---------------------------------------------------------------------------

target_include_directories(anomalon PRIVATE
    include
    ${LUA_INCLUDE_DIR}
)

# ---------------------------------------------------------------------------
# Link Dependency Libraries
# ---------------------------------------------------------------------------

target_link_libraries(anomalon PRIVATE
    Boost::system
    OpenSSL::SSL
    OpenSSL::Crypto
    ${LUA_LIBRARIES}
    sol2::sol2
    Threads::Threads # Ensures standard OS thread pool library linkage (-pthread)
)

# ---------------------------------------------------------------------------
# Compiler Flags Configuration (Platform Specific)
# ---------------------------------------------------------------------------

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_ID MATCHES "GNU")
    target_compile_options(anomalon PRIVATE
        -Wall
        -Wextra
        -Wpedantic
        -O3             # Aggressive speed optimization for production
        -march=native   # Optimizes binary specifically for your host CPU architecture
    )
elseif(CMAKE_CXX_COMPILER_ID MATCHES "MSVC")
    target_compile_options(anomalon PRIVATE
        /W4             # High warning thresholds for Visual Studio
        /O2             # Maximize execution speed
        /permissive-    # Enforces strict standard C++ conformance
    )

# ---------------------------------------------------------------------------
# Post-Build Asset Copy Rule
# ---------------------------------------------------------------------------
# Automatically clones the server.lua scripts folder over to your build output

add_custom_command(TARGET anomalon POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts
    $<TARGET_FILE_DIR:anomalon>/scripts
)
```

## Tips for building this template smoothly:

* The Threads Library: Threads::Threads is explicitly injected into the build targets to tell your compiler (like GCC or Clang) to link against -pthread, which is vital since your engine explicitly creates system pools using std::thread.
* Vcpkg / Conan Integration: If you are using a C++ package manager like vcpkg to maintain Boost, OpenSSL, and sol2, make sure to append the toolchain file when generating your workspace:

```bash
cmake -B build -S . -DCMAKE_TOOLCHAIN_FILE=[path_to_vcpkg]/scripts/buildsystems/vcpkg.cmake
```
